    <script>
        // ==================== 在这里修改你的菜单 ====================
        const dishes = [
            { name: "番茄炒蛋", emoji: "🍅", desc: "经典家常" },
            { name: "红烧肉", emoji: "🥩", desc: "肥而不腻" },
            { name: "麻婆豆腐", emoji: "🌶️", desc: "麻辣鲜香" },
            { name: "糖醋排骨", emoji: "🍖", desc: "酸甜可口" },
            { name: "清蒸鱼", emoji: "🐟", desc: "鲜嫩清淡" },
            { name: "宫保鸡丁", emoji: "🍗", desc: "下饭神器" },
            { name: "蒜蓉西兰花", emoji: "🥦", desc: "健康清爽" },
            { name: "蛋炒饭", emoji: "🍚", desc: "简单美味" },
            { name: "可乐鸡翅", emoji: "🍗", desc: "甜咸适口" },
            { name: "酸辣土豆丝", emoji: "🥔", desc: "开胃爽口" },
            { name: "西红柿牛腩", emoji: "🐄", desc: "浓郁汤汁" },
            { name: "蒜蓉粉丝虾", emoji: "🦐", desc: "鲜美弹牙" },
        ];
        // ==========================================================

        let selected = new Set();
        const MAX_SELECT = 5;

        function render() {
            const grid = document.getElementById('menuGrid');
            grid.innerHTML = dishes.map((dish, i) => `
                <div class="dish-card ${selected.has(i) ? 'selected' : ''}" onclick="toggle(${i})">
                    <div class="check-mark">✓</div>
                    <span class="dish-emoji">${dish.emoji}</span>
                    <div class="dish-name">${dish.name}</div>
                    <div class="dish-desc">${dish.desc}</div>
                </div>
            `).join('');

            document.getElementById('count').textContent = selected.size;

            const tags = document.getElementById('selectedTags');
            if (selected.size === 0) {
                tags.innerHTML = '<span style="color: #b2bec3; font-size: 0.9rem;">还没有选菜哦～</span>';
            } else {
                tags.innerHTML = Array.from(selected).map(i => 
                    `<span class="tag">${dishes[i].emoji} ${dishes[i].name}</span>`
                ).join('');
            }
        }

        function toggle(index) {
            if (selected.has(index)) {
                selected.delete(index);
            } else {
                if (selected.size >= MAX_SELECT) {
                    showToast(`最多只能选 ${MAX_SELECT} 道菜哦！`);
                    return;
                }
                selected.add(index);
            }
            render();
        }

        function clearAll() {
            selected.clear();
            render();
            showToast('已重置');
        }

        function copyShare() {
            if (selected.size === 0) {
                showToast('先选几道菜吧！');
                return;
            }
            const list = Array.from(selected).map(i => `${dishes[i].emoji} ${dishes[i].name}`).join('、');
            const text = `🍽️ 今晚想吃：\n${list}\n\n快来投票吧！👆`;

            if (navigator.clipboard) {
                navigator.clipboard.writeText(text).then(() => showToast('已复制到剪贴板！'));
            } else {
                const ta = document.createElement('textarea');
                ta.value = text;
                document.body.appendChild(ta);
                ta.select();
                document.execCommand('copy');
                document.body.removeChild(ta);
                showToast('已复制到剪贴板！');
            }
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            toast.textContent = msg;
            toast.classList.add('show');
            setTimeout(() => toast.classList.remove('show'), 2000);
        }

        render();
    </script>
